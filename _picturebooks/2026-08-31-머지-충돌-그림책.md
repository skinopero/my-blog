      }, {threshold:.15});
      els.forEach(function(el){ io.observe(el); });
    } else {
      els.forEach(function(el){ el.classList.add('in-view'); });
    }
  }catch(err){
    document.querySelectorAll('[data-scene]').forEach(function(el){ el.classList.add('in-view'); });
  }
</script>
